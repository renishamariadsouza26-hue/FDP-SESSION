#Creating Data & Data Frame
import pandas as pd
data={
    "Age":[30,35,40,45,50,38],
    "Income":[50000,45000,50000,65000,70000,86000],
    "Cibil":[650,700,720,600,760,590]
}
df=pd.DataFrame(data)
#####################
print(df)
################
def decision(row):
    if row["Income"] > 30000 and row["Cibil"] > 700:
        return"Sanction"
    if row["Age"] > 55 and row["Income"] > 30000 and row["Cibil"] > 700:
            return "Appraisal"
        
    else:
        return "Reject"

df["Decision"] = df.apply(decision, axis=1)
################
print(df)
################
df["Annual Income"]=df["Income"]*12
################
print(df)

#Graphical 
#Scatterplot 
import matplotlib.pyplot as plt
import seaborn as sns

sns.scatterplot(x='Income', y='Cibil', hue='Decision', data=df)
plt.title("Income vs CIBIL Score")
plt.show()

#Bar Chart
df['Decision'].value_counts().plot(kind='bar')
plt.title("Loan Decision Count")
plt.xlabel("Decision")
plt.ylabel("Count")
plt.show()

#Boxplot
sns.boxplot(x='Decision', y='Income', data=df)
plt.title("Income Distribution by Decision")
plt.show()

#Heatmap
corr = df.corr(numeric_only=True)

sns.heatmap(corr, annot=True, cmap='coolwarm')
plt.title("Correlation Matrix")
plt.show()

sns.scatterplot(
    x='Income',
    y='Cibil',
    hue='Decision',
    data=df,
    palette={'Sanction': 'green', 'Reject': 'red'}
)
plt.title("Income vs CIBIL Score")
plt.show()
